cmake_minimum_required(VERSION 3.16)
project(parallel_tests C)
enable_testing()
set(ESP_PID_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../managed_components/esp_pid" CACHE PATH "esp_pid checkout")
set(ESP_PR_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../components/esp_pr" CACHE PATH "esp_pr checkout")
set(ESP_ANTI_WINDUP_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../components/esp_anti_windup" CACHE PATH "esp_anti_windup checkout")
add_executable(test_parallel test_parallel.c ../../app_parallel.c ../../app_control.c ${ESP_PR_SOURCE_DIR}/esp_pr.c ${ESP_PID_SOURCE_DIR}/esp_pid.c ${ESP_ANTI_WINDUP_SOURCE_DIR}/esp_anti_windup.c)
target_include_directories(test_parallel PRIVATE ../telemetry/stubs ../.. ${ESP_PR_SOURCE_DIR}/include ${ESP_PID_SOURCE_DIR}/include ${ESP_ANTI_WINDUP_SOURCE_DIR}/include)
target_compile_features(test_parallel PRIVATE c_std_11)
target_compile_options(test_parallel PRIVATE -O2 -Wall -Wextra -Werror -UNDEBUG)
target_link_libraries(test_parallel PRIVATE m)
add_test(NAME parallel_control COMMAND test_parallel)
