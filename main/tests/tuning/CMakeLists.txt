cmake_minimum_required(VERSION 3.16)
project(tuning_tests C)
enable_testing()
set(ESP_PID_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../managed_components/esp_pid" CACHE PATH "esp_pid checkout")
set(ESP_PR_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../components/esp_pr" CACHE PATH "esp_pr checkout")
set(ESP_ANTI_WINDUP_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../components/esp_anti_windup" CACHE PATH "esp_anti_windup checkout")
add_executable(test_tuning test_tuning.c ../../app_tuning.c ../../app_control.c ../../app_parallel.c ${ESP_PID_SOURCE_DIR}/esp_pid.c ${ESP_ANTI_WINDUP_SOURCE_DIR}/esp_anti_windup.c ${ESP_PR_SOURCE_DIR}/esp_pr.c)
target_include_directories(test_tuning PRIVATE stubs ../telemetry/stubs ../.. ${ESP_PR_SOURCE_DIR}/include ${ESP_PID_SOURCE_DIR}/include ${ESP_ANTI_WINDUP_SOURCE_DIR}/include)
target_compile_features(test_tuning PRIVATE c_std_11)
target_compile_options(test_tuning PRIVATE -O2 -Wall -Wextra -Wno-unused-variable -UNDEBUG)
target_link_libraries(test_tuning PRIVATE m)
add_test(NAME bounded_tuning COMMAND test_tuning)
