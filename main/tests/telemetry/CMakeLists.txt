cmake_minimum_required(VERSION 3.16)
project(telemetry_handoff_tests C)
enable_testing()
set(ESP_PID_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../managed_components/esp_pid" CACHE PATH "esp_pid checkout")
set(ESP_PR_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../components/esp_pr" CACHE PATH "esp_pr checkout")
set(ESP_ANTI_WINDUP_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../components/esp_anti_windup" CACHE PATH "esp_anti_windup checkout")
find_package(Threads REQUIRED)
add_executable(test_telemetry ../../app_control.c ../../app_parallel.c ${ESP_PID_SOURCE_DIR}/esp_pid.c ${ESP_ANTI_WINDUP_SOURCE_DIR}/esp_anti_windup.c ${ESP_PR_SOURCE_DIR}/esp_pr.c test_telemetry.c)
target_include_directories(test_telemetry PRIVATE stubs ../.. ${ESP_PR_SOURCE_DIR}/include ${ESP_PID_SOURCE_DIR}/include ${ESP_ANTI_WINDUP_SOURCE_DIR}/include)
target_compile_features(test_telemetry PRIVATE c_std_11)
target_compile_options(test_telemetry PRIVATE -O2 -Wall -Wextra -Werror -UNDEBUG)
target_link_libraries(test_telemetry PRIVATE Threads::Threads m)
add_test(NAME telemetry_handoff COMMAND test_telemetry)
set_tests_properties(telemetry_handoff PROPERTIES TIMEOUT 15)

add_executable(test_signal_report ../../app_control.c ../../app_parallel.c ${ESP_PID_SOURCE_DIR}/esp_pid.c ${ESP_ANTI_WINDUP_SOURCE_DIR}/esp_anti_windup.c ${ESP_PR_SOURCE_DIR}/esp_pr.c test_signal_report.c)
target_include_directories(test_signal_report PRIVATE stubs ../.. ${ESP_PR_SOURCE_DIR}/include ${ESP_PID_SOURCE_DIR}/include ${ESP_ANTI_WINDUP_SOURCE_DIR}/include)
target_compile_features(test_signal_report PRIVATE c_std_11)
target_compile_definitions(test_signal_report PRIVATE CONFIG_APP_RMS_LOG_ENABLE=1)
target_compile_options(test_signal_report PRIVATE -O2 -Wall -Wextra -Wno-unused-variable -Werror -UNDEBUG)
target_link_libraries(test_signal_report PRIVATE m)
add_test(NAME signal_rms_reporting COMMAND test_signal_report)
